// ---------------- POSTS / FEED ----------------
match /posts/{postId} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
  allow update, delete: if request.auth != null && resource.data.uid == request.auth.uid;
  match /likes/{uid} {
    allow read: if request.auth != null;
    allow create, delete: if request.auth != null && uid == request.auth.uid;
  }
  match /comments/{commentId} {
    allow read: if request.auth != null;
    allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
    allow update, delete: if request.auth != null && resource.data.uid == request.auth.uid;
  }
}

// ---------------- SAVED POSTS ----------------
match /users/{userId}/savedPosts/{postId} {
  allow read: if request.auth != null && request.auth.uid == userId;
  allow create, update, delete: if request.auth != null && request.auth.uid == userId;
}

// ---------------- NOTIFICATIONS ----------------
match /users/{userId}/notifications/{notificationId} {
  allow read, update, delete: if request.auth != null && request.auth.uid == userId;
  // Cross-user notification creation is intentionally allowed for this client-only build.
  // Production: move notification creation to trusted backend/Cloud Functions.
  allow create: if request.auth != null;
}

// ---------------- STORIES ----------------
match /stories/{storyId} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
  allow update, delete: if request.auth != null && resource.data.uid == request.auth.uid;
  match /views/{uid} {
    allow read: if request.auth != null;
    allow create, update: if request.auth != null && uid == request.auth.uid;
    allow delete: if request.auth != null && uid == request.auth.uid;
  }
}

// ---------------- COMMUNITIES / GROUPS / CHANNELS ----------------
match /groups/{groupId} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && request.resource.data.ownerUid == request.auth.uid;
  allow update, delete: if request.auth != null && resource.data.ownerUid == request.auth.uid;
  match /members/{uid} {
    allow read: if request.auth != null;
    allow create, update: if request.auth != null && uid == request.auth.uid;
    allow delete: if request.auth != null && (uid == request.auth.uid || get(/databases/$(database)/documents/groups/$(groupId)).data.ownerUid == request.auth.uid);
  }
}

// ---------------- NOTES ----------------
